///////////////////////////////////////
// MODULE 1 – INTERACTION : DIALOGUE //
///////////////////////////////////////

fonction lancerDialogue(joueur, PNJ):
    si joueur.PHY == 0 ou joueur.MEN == 0 → return "Incapacité"
    si joueur.dialogueFaitDansTranche(PNJ) → return "Tranche horaire dépassée"
    si !PNJ.localisé(joueur.heure) → return "Lieu inconnu"

    cartes ← sélectionnerCartes(["Dialogue", "Abstraite", "Objet/PNJ/Lieu"])
    bonus ← calculerModificateurs(joueur, cartes)
    score_joueur ← arrondi((PHY + 2×MEN + HEA) / 4) + bonus
    score_PNJ ← PNJ.baseDialogueScore() - PNJ.malusMensonge() - PNJ.Wan

    jet_joueur ← D10()
    jet_PNJ ← D10()
    
    analyserRésultatDialogue(joueur, PNJ, jet_joueur, score_joueur, jet_PNJ, score_PNJ)
    joueur.temps += 2h


///////////////////////////////////////
// MODULE 2 – INTERACTION : COMBAT   //
///////////////////////////////////////

fonction lancerCombat(joueur, PNJ):
    cartes ← sélectionnerCartes(["Combat", "Spéciale", "Abstraite"])
    score_joueur ← arrondi((2×PHY + MEN + HEA) / 4) + calculerModificateurs(joueur, cartes)
    score_PNJ ← arrondi((2×PNJ.PHY + PNJ.MEN + PNJ.HEA) / 4) - PNJ.Wan

    jet_joueur ← D10()
    jet_PNJ ← D10()

    analyserRésultatCombat(joueur, PNJ, jet_joueur, score_joueur, jet_PNJ, score_PNJ)
    joueur.temps += 2h


////////////////////////////////////////////
// MODULE 3 – INTERACTION : INVESTIGATION //
////////////////////////////////////////////

fonction lancerInvestigation(joueur, lieu):
    si lieu.visitéAujourd’hui → return "Déjà investigué"
    si lieu.locked et !hasFouille(joueur) → return "Fermé"

    cartes ← sélectionnerCartes(["Investigation", "Spéciale", "Abstraite"])
    score ← arrondi((PHY + MEN + HEA) / 3) + calculerModificateurs(joueur, cartes)
    jet ← D10()

    analyserRésultatInvestigation(joueur, lieu, jet, score)
    joueur.temps += 2h


///////////////////////////////////////
// MODULE 4 – SYSTÈME DE TEMPS       //
///////////////////////////////////////

fonction avancerTemps(joueur, heures):
    joueur.temps += heures
    joueur.fatigue += heures
    actualiserTrancheHoraire()

    si joueur.fatigue > 16:
        joueur.malusDé += joueur.fatigue - 16

    si joueur.fatigue >= 24:
        joueur.dormir(12)
        joueur.HEA -= 1


////////////////////////////////////////
// MODULE 5 – SOMMEIL ET RÉCUPÉRATION //
////////////////////////////////////////

fonction dormir(joueur, durée):
    effets = tableauSommeil[durée]
    joueur.HEA += effets["HEA"]
    joueur.Emo += effets["Emo"]
    joueur.bonusDé = effets["BonusDé"]
    joueur.fatigue = 0
    joueur.aDormitAujourd’hui = vrai
    joueur.Add -= 1


///////////////////////////////////////
// MODULE 6 – ADDICTION              //
///////////////////////////////////////

fonction consommerDose(joueur, nb):
    si joueur.dosesConsomméesAujourd’hui + nb > 2:
        return "Overdose !"
    joueur.Add += nb
    joueur.dosesConsomméesAujourd’hui += nb

fonction vérifierAddiction(joueur):
    si joueur.Add == -3 ou joueur.Add == 3:
        jet ← D10()
        si jet > joueur.HEA:
            joueur.mourir()
        sinon:
            joueur.HEA -= 1
            joueur.Add = signe(joueur.Add) * 2


///////////////////////////////////////
// MODULE 7 – SYSTÈME DE SANTÉ       //
///////////////////////////////////////

fonction vérifierSanté(joueur):
    si joueur.HEA <= 0:
        joueur.mourir()
    si joueur.PHY == 0 ou joueur.MEN == 0:
        joueur.reposComplet()


///////////////////////////////////////
// MODULE 8 – PARANOÏA               //
///////////////////////////////////////

fonction vérifierParanoia(joueur):
    if joueur.compteurParano >= seuil:
        joueur.devenirParano()
        joueur.modeParanoActif = vrai
        joueur.tempsRestant = 7 jours
        joueur.Emo = 0
        joueur.Add = 0
        joueur.PHY += 2
        joueur.MEN -= 1
        joueur.HEA -= 1


///////////////////////////////////////////
// MODULE 9 – ÉVOLUTION CARACTÉRISTIQUES //
///////////////////////////////////////////

fonction appliquerCritique(joueur, typeJet):
    stat = choixAléatoire(caractéristiquesUtilisées)
    si typeJet == "réussite":
        stat += 1
    sinon:
        stat -= 1
    si stat == HEA:
        HEAnom = HEA //  Pas sûr de la formule... Je me demande si ça ne serait pas l'inverse.


///////////////////////////////////////
// MODULE 10 – SYSTÈME DE CARTES     //
///////////////////////////////////////

fonction gérerCartes(joueur, cartesRéussies):
    pour chaque carte dans cartesRéussies:
        si carte ∉ PileMaîtrise:
            ajouterCartePileMaîtrise(joueur, carte)

    si nb_cartes > 10:
        joueur.choisirCartesÀRetirer()


///////////////////////////////////////
// MODULE 11 – SYSTÈME D’ENQUÊTES    //
///////////////////////////////////////

fonction validerEnquête(joueur, enquête):
    si !enquête.aTousLesLiensCorrects():
        appliquerMalus(joueur)
        return "Échec"
    si enquête.mobileIncorrect:
        donnerMoinsBonus(joueur)
    sinon:
        donnerGrosBonus(joueur)
    enquête.terminée = vrai


///////////////////////////////////////
// MODULE 12 – GESTION DES PNJs      //
///////////////////////////////////////

fonction évolutionPNJ(pnj):
    si jet(1,10) == 1:
        stat = aléatoire([PHY, MEN, HEAnom])
        stat += aléatoire([-1, +1])

fonction interactionAvecPNJ(joueur, pnj):
    mettreÀJourRep(pnj, joueur)
    calculerRisqueAgression(pnj, joueur)

    si pnj.Wan == 5:
        pnj.emprisonner()
