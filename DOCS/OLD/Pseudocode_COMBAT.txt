ALGORITHME COMPLET – INTERACTION DE COMBAT (PSEUDO-CODE LOGIQUE)

// === 1. DÉCLENCHEMENT DU COMBAT ===
// SI (Combat initié par Joueur) ALORS
//     afficher("Tu ne peux pas provoquer un Combat.")
//     TERMINER

SI (Combat initié par PNJ) ALORS
    afficher("Agression en cours !")
    joueur_peut_fuir ← vrai

// === 2. FUITE AVANT COMBAT (Optionnelle) ===
SI joueur_peut_fuir ALORS
    proposerOption("Fuir ?")
    SI joueur_accepte_fuite ALORS
        cartes_fuite ← sélectionnerCarte("Run Away" + Abstraite)
        modif ← calculerModificateurs(cartes_fuite + Emo, Add, Wan, Veille, Sommeil, etc.)
        score_fuite ← arrondi((PHY + MEN + HEA) / 3) + modif
        jet_fuite ← lancerDé(1,10)

        SI jet_fuite == 1 ALORS
            modifierEmo(+1 vers 0)
            afficher("Fuite critique réussie !")
            TERMINER

        SI jet_fuite ≤ score_fuite ALORS
            afficher("Tu t’échappes.")
            TERMINER

        SI jet_fuite == 10 ALORS
            modifierHEA(-1)
            modifierEmo(-1 Min/Max)
            afficher("Échec critique de la fuite ! Tu es blessé.")
            passerÀCombat

        SINON
            modifierEmo(-1 Min/Max)
            afficher("Tu rates ta fuite, le combat commence !")

// === 3. PRÉPARATION DU COMBAT ===
cartes_combat ← sélectionnerCartes(1 Combat, 1 Combat Spécial, 1 Abstraite)
bonus_malus ← calculerBonusMalus(cartes_combat + Emo, Add, Wan, Rep, Veille, Sommeil, Lieu)

// === 4. CALCULS DES SCORES ===
score_base_joueur ← arrondi((PHY×2 + MEN + HEA) / 4)
score_total_joueur ← score_base_joueur + bonus_malus

score_base_pnj ← arrondi((PHY_PNJ×2 + MEN_PNJ + HEA_PNJ) / 4)
score_total_pnj ← score_base_pnj - Wan_PNJ

jet_joueur ← lancerDé(1,10)
jet_pnj ← lancerDé(1,10)

// === 5. RÉSOLUTION DU COMBAT ===
SI jet_joueur == 1 ALORS
    modifierEmo(+1 vers 0)
    SI carte "Ne pas tuer !" utilisée ALORS
        PNJ.HEA = 1
        afficher("Tu l’as épargné, just in time !")
        allerÀFinCombat

SI jet_joueur == 10 ALORS
    modifierEmo(-1 Min/Max)
    modifierHEA(-1)
    afficher("Échec critique ! Tu te blesses.")

SI jet_pnj == 10 ALORS
    afficher("Échec critique du PNJ. Il se blesse !")
    PNJ.HEA -= 1

// === COMPARAISON DES SUCCÈS ===
réussite_joueur ← jet_joueur ≤ score_total_joueur
réussite_pnj ← jet_pnj ≤ score_total_pnj

SI les deux réussissent ALORS
    marge_joueur ← score_total_joueur - jet_joueur
    marge_pnj ← score_total_pnj - jet_pnj

    SI marge_joueur > marge_pnj ALORS
        dégâts ← marge_joueur
        PNJ.HEA -= dégâts
    SINON SI marge_pnj > marge_joueur ALORS
        dégâts ← marge_pnj
        HEA -= dégâts
    SINON
        afficher("Égalité, status quo.")

SINON SI réussite_joueur ALORS
    dégâts ← score_total_joueur - jet_joueur
    PNJ.HEA -= dégâts

SINON SI réussite_pnj ALORS
    dégâts ← score_total_pnj - jet_pnj
    HEA -= dégâts

SINON
    afficher("Échec mutuel. Rien ne se passe.")

// === 6. CONSÉQUENCES ===
SI PNJ.HEA <= 0 ALORS
    PNJ.mort = vrai
    joueur.Wan += 1
    afficher("Tu as tué le PNJ.")
    SI random(0,3) == 0 ALORS
        gagnerArgent(50±10 %)
        afficher("Tu trouves du fric sur lui.")

SINON
    PNJ.inaccessibleJusquÀ(demain)
    PNJ.Rep = -3

SI joueur.HEA <= 0 ALORS
    afficher("Tu es mort.")
    TERMINER

// === 7. PROGRESSION ===
SI jet_joueur == 1 OU jet_joueur == 10 ALORS
    stat_utilisée ← aléatoire([PHY, MEN, HEA])
    SI jet == 1 → stat_utilisée += 1
    SI jet == 10 → stat_utilisée -= 1

// === 8. MISE À JOUR TEMPS ET INTERACTIONS ===
temps += 2h
autres_PNJs_sur_place → fuient et inaccessibles 1 jour
PNJ reprendra emploi du temps après interaction

// === 9. PRISON SI WAN = 5 ===
SI PNJ.Wan >= 5 ALORS
    PNJ.état = "Prison"
    PNJ.accessibleSeulementMatinAprèsMidi = vrai

FIN
