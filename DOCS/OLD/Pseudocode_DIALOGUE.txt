ALGORITHME COMPLET – INTERACTION DE DIALOGUE (PSEUDO-CODE LOGIQUE)

// === DÉMARRAGE ===
SI (PHY == 0 OU MEN == 0) ALORS
    afficher("Impossible d'interagir, vous devez vous reposer.")
    TERMINER

SI (Déjà dialogué avec ce PNJ dans cette tranche horaire) ALORS
    afficher("Vous devez attendre la prochaine tranche horaire.")
    TERMINER

SI (Emploi du temps PNJ inconnu) ALORS
    afficher("Je ne sais pas où se trouve cette personne en ce moment.")
    TERMINER

SI (PNJ absent de l'endroit prévu) ALORS
    ajouterHeure(1)
    afficher("Personne ici... étrange.")
    TERMINER

// === PRÉPARATION DU DIALOGUE ===
cartes_jouées ← sélectionnerCartes()  // max 1 Dialogue + 1 Abstraite + 1 Objet/PNJ/Lieu

SI (Aucune carte) ALORS
    afficher("Il faut au moins une carte pour interagir.")
    TERMINER

SI (nbCartes > 3) ALORS
    afficher("Trop de cartes jouées.")
    TERMINER

bonus_malus ← calculerBonusMalus(cartes_jouées, joueur)

// === CALCUL DU SCORE DU JOUEUR ===
score_base_joueur ← arrondi((PHY + 2×MEN + HEA) / 4)
score_total_joueur ← score_base_joueur + bonus_malus

// === CALCUL DU SCORE DU PNJ ===
score_base_pnj ← arrondi((PHY_PNJ + 2×MEN_PNJ + HEA_PNJ) / 4)
malus_pnj ← Wan_PNJ + malusMensonge(PNJ)
score_total_pnj ← score_base_pnj - malus_pnj

// === JETS DE DÉ ===
jet_joueur ← lancerDé(1, 10)
jet_pnj ← lancerDé(1, 10)

réussite_joueur ← jet_joueur ≤ score_total_joueur
réussite_pnj ← jet_pnj ≤ score_total_pnj

// === ANALYSE DES CAS CRITIQUES ===
SI (jet_joueur == 1) ALORS
    afficher("Réussite critique !")
    modifierEmo(+1 vers 0)
    info ← révélerInfoCachée(PNJ)
    allerÀFinInteraction()

SI (jet_joueur == 10) ALORS
    afficher("Échec critique...")
    modifierEmo(-1 Min/Max)
    modifierRep(PNJ, -1)
    allerÀFinInteraction()

// === COMPARAISON EN CAS DE DOUBLE RÉUSSITE ===
SI (réussite_joueur ET réussite_pnj) ALORS
    marge_joueur ← score_total_joueur - jet_joueur
    marge_pnj ← score_total_pnj - jet_pnj

    SI (marge_joueur > marge_pnj) ALORS
        infos ← tirerInfos(PNJ, cartes_jouées)
    SINON SI (marge_joueur < marge_pnj) ALORS
        afficher("Le PNJ résiste à vos questions.")
    SINON
        afficher("Égalité : rien ne se passe.")

// === CAS AUTRES ===
SINON SI (réussite_joueur) ALORS
    infos ← tirerInfos(PNJ, cartes_jouées)
    SI (marge_joueur ≥ 3) ALORS
        ajouterInfoCachée(PNJ)
SINON
    afficher("Le PNJ vous baratine.")
    // Rien obtenu

// === ÉVOLUTION DES STATS (si crit) ===
SI (jet_joueur == 1 OU jet_joueur == 10) ALORS
    stat_utilisée ← choisirAléatoirement([PHY, MEN, HEA])
    SI (jet == 1) ALORS stat_utilisée += 1
    SI (jet == 10) ALORS stat_utilisée -= 1
    HEAnom = HEA si modifié

// === GESTION DES INFOS/MENSONGES ===
POUR CHAQUE info DANS infos
    SI info_estIncriminante ALORS
        taux_mensonge ← calculerTauxMensonge(PNJ, MEN)
    SINON
        taux_mensonge ← calculerTauxMensongeSimple(PNJ, MEN)

    SI (rand() < taux_mensonge) ALORS
        info.estFausse ← vrai
        PNJ.malus_mensonge += 1

    ajouterInfoFicheEnquête(info)

// === POST-INTERACTION ===
temps += 2h
PNJ.inaccessibleJusquÀ(demain)
PNJ.reprendEmploiDuTemps()
carteRéussies ← filtrerCartesRéussies(cartes_jouées)

ajouterÀPileMaîtrise(carteRéussies)

// === DÉCLENCHEURS SPÉCIAUX ===
SI Add == ±3 ALORS
    testMort ← lancerDé(1, 10) ≤ HEA
    SI (testMort ÉCHOUÉ) ALORS joueur.meurt()
    SINON
        HEA -= 1
        modifierEmo(±1)
        Add = ±2

SI Emo == ±3 OU criseParanoia() ALORS
    afficherOverlay("PARANOÏA !")
    terminerInteraction()
    joueurEntreParanoia()

FIN
